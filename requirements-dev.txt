# Dev / test dependencies for the export pipeline.
#
# These are NOT in requirements.txt (keeps the base OSS install light;
# the app lazy-imports these at runtime and degrades gracefully when absent).
#
# Install locally to run the full export test suite:
#
#   pip install -r requirements-dev.txt
#   # For cairosvg (PPTX tests) you also need the system cairo library:
#   #   Linux:  apt-get install libcairo2
#   #   macOS:  brew install cairo
#
# The CI workflow installs libcairo2 via apt and then installs this file so
# that test_render_pdf.py, test_t4_render_pptx.py, and their skip guards
# never trigger (tests run rather than skip).

# PDF export backend (svglib is the pure-Python fallback for test_render_pdf.py)
svglib>=1.5
reportlab>=4.0

# PDF page-count assertions in test_render_pdf.py
pypdf>=3.0

# PPTX export (test_t4_render_pptx.py)
python-pptx>=1.0

# cairosvg — preferred PDF backend + PPTX per-widget rasteriser.
# Requires the system libcairo library (installed separately in CI).
# Listed here so `pip install -r requirements-dev.txt` pulls it in once
# the system library is present.  Pure-Python tests use svglib instead.
cairosvg>=2.7
